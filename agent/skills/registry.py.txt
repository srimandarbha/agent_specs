"""Reason-code registry.

Human source: contracts/REASON_CODES.md, section 3 tables.
Machine source: contracts/reason_codes.json, generated by tools/build_reason_registry.py.
A test fails if the two drift apart.
"""
from __future__ import annotations
import json, pathlib, re

REPO = pathlib.Path(__file__).resolve().parents[2]
MD_PATH = REPO / "contracts" / "REASON_CODES.md"
JSON_PATH = REPO / "contracts" / "reason_codes.json"
CODE = re.compile(r"^`([a-z][a-z0-9_]*)`$")
STATUS = re.compile(r"^`(supported|probable|inconclusive|not_supported)`")
UC = re.compile(r"UC\d\d")


def parse_markdown(text: str | None = None) -> list[dict]:
    text = MD_PATH.read_text() if text is None else text
    rows, in_section = [], False
    for line in text.splitlines():
        if line.startswith("## "):
            in_section = line.startswith("## 3")
            continue
        if not in_section or not line.startswith("|"):
            continue
        cells = [c.strip() for c in line.strip().strip("|").split("|")]
        if len(cells) < 4:
            continue
        m, s = CODE.match(cells[0]), STATUS.match(cells[1])
        if not (m and s):
            continue
        rows.append({
            "code": m.group(1),
            "classification": s.group(1),
            "remediation": cells[2].strip("`"),
            "description": cells[3],
            "applicable_use_cases": UC.findall(cells[4]) if len(cells) > 4 else [],
            "scope": "global" if len(cells) > 4 and "All" in cells[4] else "scoped",
        })
    return rows


def build_registry() -> dict:
    rows = parse_markdown()
    seen: dict[str, dict] = {}
    for r in rows:
        if r["code"] in seen and seen[r["code"]] != r:
            raise ValueError(f"conflicting duplicate reason code: {r['code']}")
        seen[r["code"]] = r
    return {"schema_version": "1.0", "source": "contracts/REASON_CODES.md section 3",
            "codes": sorted(seen.values(), key=lambda r: r["code"])}


def load_registry() -> dict[str, dict]:
    return {c["code"]: c for c in json.loads(JSON_PATH.read_text())["codes"]}
